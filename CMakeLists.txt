cmake_minimum_required(VERSION 3.24)
project(TypePick VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/utf-8 /W4 /EHsc)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_library(typepick_core native/src/selector.cpp)
target_include_directories(typepick_core PUBLIC native/include third_party)
if(WIN32)
  add_library(typepick_windows native/src/jev_client_win.cpp native/src/weasel_bridge_win.cpp)
  target_include_directories(typepick_windows PUBLIC third_party/librime)
  target_link_libraries(typepick_windows PUBLIC typepick_core winhttp user32 gdi32 shell32 advapi32)
  add_executable(TypePick WIN32 native/src/settings_win.cpp)
  target_link_libraries(TypePick PRIVATE typepick_windows comdlg32 ole32)
  add_executable(TypePickProbe native/src/probe_win.cpp)
  target_link_libraries(TypePickProbe PRIVATE typepick_windows)
endif()
option(BUILD_TESTING "Build TypePick tests" ON)
if(BUILD_TESTING)
  enable_testing()
  add_executable(typepick_tests native/tests/selector_tests.cpp)
  target_link_libraries(typepick_tests PRIVATE typepick_core)
  add_test(NAME selector COMMAND typepick_tests)
  add_executable(diagnostics_tests native/tests/diagnostics_tests.cpp)
  target_link_libraries(diagnostics_tests PRIVATE typepick_core)
  add_test(NAME diagnostics COMMAND diagnostics_tests)
  add_executable(personal_tests native/tests/personal_tests.cpp)
  target_link_libraries(personal_tests PRIVATE typepick_core)
  add_test(NAME personal COMMAND personal_tests)
  if(WIN32)
    add_executable(popup_layout_tests native/tests/popup_layout_tests.cpp)
    target_include_directories(popup_layout_tests PRIVATE native/include)
    target_link_libraries(popup_layout_tests PRIVATE user32)
    add_test(NAME popup_layout COMMAND popup_layout_tests)
  endif()
endif()
